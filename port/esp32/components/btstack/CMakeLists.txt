# ESP-IDF first evaluates every component in a lightweight, script-mode pass
# to discover only component dependencies. There is no configured C compiler
# in that pass, so defer btstack_config.h preprocessing and source selection
# to the normal component evaluation below.
if(CMAKE_BUILD_EARLY_EXPANSION)
    # Dependencies must be registered here as well: ESP-IDF uses this pass to
    # construct the component graph, including propagated SDK include paths.
    if(IDF_VERSION_MAJOR GREATER 5 OR
       (IDF_VERSION_MAJOR EQUAL 5 AND IDF_VERSION_MINOR GREATER_EQUAL 3))
        set(BTSTACK_ESP32_EARLY_DRIVER_REQUIRES
            esp_driver_gpio
            esp_driver_uart
            esp_driver_i2s
            esp_driver_usb_serial_jtag)
    else()
        set(BTSTACK_ESP32_EARLY_DRIVER_REQUIRES driver)
    endif()
    idf_component_register(
        PRIV_REQUIRES nvs_flash bt lwip vfs ${BTSTACK_ESP32_EARLY_DRIVER_REQUIRES}
        REQUIRES btstack_config)
    return()
endif()

# sdkconfig.h is generated by ESP-IDF before components are configured. Feed
# it, together with ESP-IDF's version header, to the normal C preprocessor so
# ENABLE_CLASSIC/ENABLE_BLE follow the selected target automatically.
btstack_declare_port(esp32)
btstack_collect_port_configuration_macros(BTSTACK_ESP32_CONFIGURATION_MACROS
    CONFIG "${CMAKE_CURRENT_LIST_DIR}/../btstack_config/include/btstack_config.h"
    INCLUDE_DIRECTORIES
        "${CMAKE_BINARY_DIR}/config"
        "${IDF_PATH}/components/esp_common/include")
# ESP-IDF already reconfigures after sdkconfig changes; retain this explicit
# dependency as well because its generated header directly controls the
# preprocessor result consumed by the resolver.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_BINARY_DIR}/config/sdkconfig.h")
btstack_set_port_configuration_macros(esp32 ${BTSTACK_ESP32_CONFIGURATION_MACROS})

if(CONFIG_BTSTACK_AUDIO)

    # btstack_port_esp32.c installs it whenever this Kconfig option is set.

        CAPABILITY audio
        PUBLIC_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
endif()

# The port itself always needs its run loop, controller chipset adapter, and
# the persistent databases guarded in btstack_port_esp32.c. The latter two
# are pruned automatically for LE-only and Classic-only controller configs.
set(BTSTACK_ESP32_BASE_GROUPS
    core
    chipset-zephyr
    le-device-db-tlv
    link-key-db-tlv)
# A static random address is only queried for the optional external Zephyr
# controller transport. The internal ESP VHCI controller needs no BLE source
# baseline beyond the selected application and LE device database.
if(CONFIG_BTSTACK_EXTERNAL_UART_BT)
    list(APPEND BTSTACK_ESP32_BASE_GROUPS ble)
endif()
btstack_filter_groups_for_port(BTSTACK_ESP32_ACTIVE_BASE_GROUPS
    PORT esp32
    GROUPS ${BTSTACK_ESP32_BASE_GROUPS})

    GROUPS ${BTSTACK_ESP32_EXAMPLE_GROUPS} ${BTSTACK_ESP32_ACTIVE_BASE_GROUPS})

    GROUPS ${BTSTACK_ESP32_GROUPS})

# Populate example-info for every declared example, not just the one ESP-IDF
# is compiling. This mirrors the other migrated ports and makes the Kconfig
# result inspectable without rebuilding each example.
foreach(example IN LISTS BTSTACK_ESP32_EXAMPLES)
    btstack_resolve_example_for_port(esp32 "${example}" example_groups example_configuration)
    btstack_resolve_groups(example_all_groups
        GROUPS ${example_groups} ${BTSTACK_ESP32_ACTIVE_BASE_GROUPS})
    btstack_collect_sources(example_sources GROUPS ${example_all_groups})
    foreach(source IN LISTS btstack_example_${example}_sources)
        list(APPEND example_sources "${BTSTACK_ROOT}/example/${source}")
    endforeach()
    list(APPEND example_sources
        "${BTSTACK_ROOT}/platform/freertos/btstack_run_loop_freertos.c"
        "${CMAKE_CURRENT_LIST_DIR}/btstack_port_esp32.c"
        "${CMAKE_CURRENT_LIST_DIR}/btstack_stdio_esp32.c"
        "${CMAKE_CURRENT_LIST_DIR}/btstack_tlv_esp32.c")
    list(REMOVE_DUPLICATES example_sources)
    list(SORT example_sources)
    string(REPLACE "-" "_" example_suffix "${example}")
    set_property(GLOBAL PROPERTY "btstack_info_esp32_${example_suffix}_sources" "${example_sources}")
    set(example_include_dirs ${include_dirs})
    list(REMOVE_DUPLICATES example_include_dirs)
    list(SORT example_include_dirs)
    set_property(GLOBAL PROPERTY "btstack_info_esp32_${example_suffix}_include_directories" "${example_include_dirs}")
endforeach()
